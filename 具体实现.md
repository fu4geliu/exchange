gateway-service        # 端口 9080，唯一对外暴露的入口
customer-service       # 端口 9081，内网，不对外
trade-service          # 端口 9082，内网，不对外
market-service         # 端口 9083，内网，不对外
common                 # 公共模块：JWT工具类、统一响应体等

项目结构
financial-system/                    ← 父 Maven 项目（只有 pom.xml）
│
├── pom.xml                          ← 父 pom，管理所有依赖版本
│
├── common/                          ← 公共模块，不是 Spring Boot
│   └── src/main/java/
│       ├── dto/                     # 统一响应体 Result<T>
│       ├── utils/JwtUtil.java       # JWT 工具类
│       └── exception/              # 全局异常定义
│
├── gateway-service/                 ← Spring Boot (Spring Cloud Gateway)
│   └── src/main/java/
│       └── filter/JwtAuthFilter    # 全局鉴权过滤器
│
├── customer-service/                ← Spring Boot (Web + MyBatis)
│   └── src/main/java/
│       ├── controller/             # 开户、登录接口
│       ├── service/
│       └── mapper/
│
├── trade-service/                   ← Spring Boot (Web + MyBatis)
│   └── src/main/java/
│       ├── controller/             # 委托、撤单、成交接口
│       ├── service/
│       └── mapper/
│
└── market-service/                  ← Python FastAPI（独立项目，不在Maven内）
    ├── main.py
    └── requirements.txt

frontend/
├── src/
│   ├── api/                  # 所有接口调用
│   │   ├── customer.js       # 开户相关
│   │   ├── trade.js          # 委托/撤单/成交
│   │   └── market.js         # 行情查询
│   │
│   ├── views/                # 页面
│   │   ├── Login.vue         
│   │   ├── customer/
│   │   │   └── Register.vue  # 客户开户录入
│   │   ├── trade/
│   │   │   ├── Entrust.vue   # 委托下单
│   │   │   ├── Cancel.vue    # 撤单
│   │   │   └── Deal.vue      # 成交录入
│   │   └── market/
│   │       └── Market.vue    # 行情展示
│   │
│   ├── router/index.js       # 路由（含鉴权拦截）
│   ├── store/                # Pinia，存 JWT token
│   └── utils/
│       └── request.js        # axios 封装，自动带 token
