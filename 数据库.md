DROP DATABASE IF EXISTS exchange;
CREATE DATABASE exchange
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_0900_ai_ci;
USE exchange;
SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- =========================
-- 1. 字典表
-- =========================
DROP TABLE IF EXISTS dict_item;
DROP TABLE IF EXISTS dict_type;

CREATE TABLE dict_type (
    dict_code        VARCHAR(20)  NOT NULL COMMENT '字典标识',
    dict_name        VARCHAR(50) NOT NULL COMMENT '字典名称',
    PRIMARY KEY (dict_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='字典表';

CREATE TABLE dict_item (
    dict_code        VARCHAR(20)  NOT NULL COMMENT '字典标识',
    item_code        VARCHAR(10)  NOT NULL COMMENT '字典项',
    item_name        VARCHAR(50) NOT NULL COMMENT '字典项名称',
    sort_no          INT          NOT NULL DEFAULT 0 COMMENT '排序号',
    PRIMARY KEY (dict_code, item_code),
    CONSTRAINT fk_dict_item_type
        FOREIGN KEY (dict_code) REFERENCES dict_type(dict_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='字典项表';

-- =========================
-- 2. 客户模板表
-- =========================
DROP TABLE IF EXISTS customer_template;
CREATE TABLE customer_template (
    cuacct_cls       VARCHAR(10)   NOT NULL COMMENT '资产账户类别(如 0/1/2/3)',
    cash_balance     DECIMAL(18,2) NOT NULL DEFAULT 0 COMMENT '资金余额',
    stk_code_1       VARCHAR(20)   DEFAULT NULL COMMENT '证券1',
    stk_balance_1    BIGINT        NOT NULL DEFAULT 0 COMMENT '证券余额1',
    stk_code_2       VARCHAR(20)   DEFAULT NULL COMMENT '证券2',
    stk_balance_2    BIGINT        NOT NULL DEFAULT 0 COMMENT '证券余额2',
    stk_code_3       VARCHAR(20)   DEFAULT NULL COMMENT '证券3',
    stk_balance_3    BIGINT        NOT NULL DEFAULT 0 COMMENT '证券余额3',
    PRIMARY KEY (cuacct_cls)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客户模板表';

-- =========================
-- 3. 证券信息表
-- =========================
DROP TABLE IF EXISTS security_info;
CREATE TABLE security_info (
    market           VARCHAR(10)   NOT NULL COMMENT '市场(引用 MARKET 字典项：0/1)',
    stk_code         VARCHAR(20)   NOT NULL COMMENT '证券代码',
    stk_name         VARCHAR(100)  NOT NULL COMMENT '证券名称',
    stk_cls          VARCHAR(10)   NOT NULL COMMENT '证券类别(如 S0)',
    stamp_tax_rate   DECIMAL(10,6) NOT NULL DEFAULT 0 COMMENT '印花税率',
    PRIMARY KEY (market, stk_code),
    UNIQUE KEY uk_security_code (stk_code),
    KEY idx_security_cls (stk_cls)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='证券信息表';

-- =========================
-- 4. 行情信息表
-- =========================
DROP TABLE IF EXISTS market_quote;
CREATE TABLE market_quote (
    stk_code          VARCHAR(20)   NOT NULL COMMENT '证券代码',
    current_price     DECIMAL(18,4) NOT NULL COMMENT '当前价格',
    prev_close_price  DECIMAL(18,4) NOT NULL COMMENT '昨收价',
    upper_limit_price DECIMAL(18,4) NOT NULL COMMENT '涨停价格',
    lower_limit_price DECIMAL(18,4) NOT NULL COMMENT '跌停价格',
    update_time       DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (stk_code),
    CONSTRAINT fk_market_quote_security
        FOREIGN KEY (stk_code) REFERENCES security_info(stk_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='行情信息表';

-- =========================
-- 5. 佣金信息表
-- =========================
DROP TABLE IF EXISTS commission_rate;
CREATE TABLE commission_rate (
    cuacct_cls       VARCHAR(5)   NOT NULL COMMENT '资产账户类别',
    market           VARCHAR(5)   NOT NULL COMMENT '市场',
    stk_cls          VARCHAR(10)   NOT NULL COMMENT '证券类别',
    commission_rate  DECIMAL(10,6) NOT NULL DEFAULT 0 COMMENT '佣金费率',
    PRIMARY KEY (cuacct_cls, market, stk_cls)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='佣金信息表';

-- =========================
-- 6. 用户信息表
-- =========================
DROP TABLE IF EXISTS user_info;
CREATE TABLE user_info (
    user_id           BIGINT        NOT NULL AUTO_INCREMENT COMMENT '用户ID',
    customer_code     BIGINT        NOT NULL COMMENT '客户代码（毫秒戳后7位+4位随机码）', 
    user_name         VARCHAR(50)   NOT NULL COMMENT '用户姓名',
    id_type           VARCHAR(10)   NOT NULL COMMENT '证件类型(00/01/02)',
    id_no             VARCHAR(50)   NOT NULL COMMENT '证件号码',
    cuacct_cls        VARCHAR(10)   NOT NULL COMMENT '资产账户类别',
    cuacct_status     VARCHAR(10)   NOT NULL DEFAULT '0' COMMENT '资产账户状态',
    created_time      DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_time      DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (user_id),
    UNIQUE KEY uk_user_id_no (id_no),
    UNIQUE KEY uk_user_customer_code (customer_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户信息表';

-- =========================
-- 7. 客户账户表
-- =========================
DROP TABLE IF EXISTS customer_account;
CREATE TABLE customer_account (
    account_id        BIGINT        NOT NULL AUTO_INCREMENT COMMENT '账户ID',
    customer_code     BIGINT        NOT NULL UNIQUE COMMENT '客户代码',
    currency          VARCHAR(10)   NOT NULL DEFAULT '0' COMMENT '币种(0-人民币/1-港币/2-美元)',
    cash_balance      DECIMAL(18,2) NOT NULL DEFAULT 0 COMMENT '现金余额',
    frozen_cash       DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (frozen_cash >= 0) COMMENT '冻结资金',
    total_assets      DECIMAL(18,2) NOT NULL DEFAULT 0 COMMENT '总资产',
    updated_time      DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (account_id),
    CONSTRAINT fk_account_customer
        FOREIGN KEY (customer_code) REFERENCES user_info(customer_code)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客户账户表';

-- =========================
-- 8. 客户持仓表
-- =========================
DROP TABLE IF EXISTS customer_position;
CREATE TABLE customer_position (
    position_id        BIGINT        NOT NULL AUTO_INCREMENT COMMENT '持仓ID',
    customer_code      BIGINT        NOT NULL COMMENT '客户代码',
    market             VARCHAR(10)   NOT NULL COMMENT '市场',
    stk_code           VARCHAR(20)   NOT NULL COMMENT '证券代码',
    stk_name           VARCHAR(100)  NOT NULL COMMENT '证券名称',
    hold_qty           BIGINT        NOT NULL DEFAULT 0 CHECK (hold_qty >= 0) COMMENT '持仓数量',
    available_qty      BIGINT        NOT NULL DEFAULT 0 CHECK (available_qty >= 0) COMMENT '可卖数量',
    frozen_qty         BIGINT        NOT NULL DEFAULT 0 CHECK (frozen_qty >= 0) COMMENT '冻结数量',
    avg_cost_price     DECIMAL(18,4) NOT NULL DEFAULT 0 CHECK (avg_cost_price >= 0) COMMENT '成本价',
    updated_time       DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (position_id),
    UNIQUE KEY uk_position_user_stock (customer_code, market, stk_code), 
    KEY idx_position_customer_code (customer_code), 
    CONSTRAINT fk_position_customer
        FOREIGN KEY (customer_code) REFERENCES user_info(customer_code) 
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_position_security
        FOREIGN KEY (market, stk_code) REFERENCES security_info(market, stk_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客户持仓表';

-- =========================
-- 9. 委托表
-- =========================
DROP TABLE IF EXISTS order_info;
CREATE TABLE order_info (
    order_id            BIGINT        NOT NULL AUTO_INCREMENT COMMENT '委托编号',
    customer_code       BIGINT        NOT NULL COMMENT '客户代码', 
    market              VARCHAR(10)   NOT NULL COMMENT '市场',
    stk_code            VARCHAR(20)   NOT NULL COMMENT '证券代码',
    trd_id              VARCHAR(10)   NOT NULL COMMENT '交易标识(B买入/S卖出)',
    order_price         DECIMAL(18,4) NOT NULL CHECK (order_price > 0) COMMENT '委托价格',
    order_qty           BIGINT        NOT NULL CHECK (order_qty > 0) COMMENT '委托数量',
    matched_qty         BIGINT        NOT NULL DEFAULT 0 CHECK (matched_qty >= 0) COMMENT '已成交数量',
    order_amount        DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (order_amount >= 0) COMMENT '委托金额',
    order_status        VARCHAR(10)   NOT NULL DEFAULT 'A' COMMENT '委托状态(字典ORDER_STATUS)',
    is_withdraw         VARCHAR(10)   NOT NULL DEFAULT 'F' COMMENT '撤单标志(字典IS_WITHDRAW)',
    remark              VARCHAR(500)  DEFAULT NULL COMMENT '备注',
    entrust_time        DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '委托时间',
    updated_time        DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (order_id),
    KEY idx_order_customer_code (customer_code), 
    KEY idx_order_stock (market, stk_code),
    KEY idx_order_status (order_status),
    CONSTRAINT fk_order_customer
        FOREIGN KEY (customer_code) REFERENCES user_info(customer_code) 
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_order_security
        FOREIGN KEY (market, stk_code) REFERENCES security_info(market, stk_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='委托表';

-- =========================
-- 10. 撤单表
-- =========================
DROP TABLE IF EXISTS withdraw_info;
CREATE TABLE withdraw_info (
    withdraw_id         BIGINT        NOT NULL AUTO_INCREMENT COMMENT '撤单编号',
    order_id            BIGINT        NOT NULL COMMENT '原委托编号',
    customer_code       BIGINT        NOT NULL COMMENT '客户代码', 
    withdraw_reason     VARCHAR(500)  DEFAULT NULL COMMENT '撤单原因',
    withdraw_time       DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '撤单时间',
    PRIMARY KEY (withdraw_id),
    KEY idx_withdraw_order (order_id),
    KEY idx_withdraw_customer_code (customer_code),
    CONSTRAINT fk_withdraw_order
        FOREIGN KEY (order_id) REFERENCES order_info(order_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_withdraw_customer
        FOREIGN KEY (customer_code) REFERENCES user_info(customer_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='撤单表';

-- =========================
-- 11. 成交表
-- =========================
DROP TABLE IF EXISTS trade_info;
CREATE TABLE trade_info (
    trade_id            BIGINT        NOT NULL AUTO_INCREMENT COMMENT '成交编号',
    order_id            BIGINT        NOT NULL COMMENT '原委托编号',
    customer_code       BIGINT        NOT NULL COMMENT '客户代码',
    market              VARCHAR(10)   NOT NULL COMMENT '市场',
    stk_code            VARCHAR(20)   NOT NULL COMMENT '证券代码',
    trd_id              VARCHAR(10)   NOT NULL COMMENT '交易标识(B买入/S卖出)',
    trade_price         DECIMAL(18,4) NOT NULL CHECK (trade_price > 0) COMMENT '成交价格',
    trade_qty           BIGINT        NOT NULL CHECK (trade_qty > 0) COMMENT '成交数量',
    trade_amount        DECIMAL(18,2) NOT NULL CHECK (trade_amount > 0) COMMENT '成交金额',
    commission_fee      DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (commission_fee >= 0) COMMENT '佣金',
    stamp_tax_fee       DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (stamp_tax_fee >= 0) COMMENT '印花税',
    total_fee           DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (total_fee >= 0) COMMENT '总费用',
    trade_time          DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '成交时间',
    PRIMARY KEY (trade_id),
    KEY idx_trade_order (order_id),
    KEY idx_trade_customer_code (customer_code), 
    KEY idx_trade_stock (market, stk_code),
    CONSTRAINT fk_trade_order
        FOREIGN KEY (order_id) REFERENCES order_info(order_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_trade_customer
        FOREIGN KEY (customer_code) REFERENCES user_info(customer_code) 
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_trade_security
        FOREIGN KEY (market, stk_code) REFERENCES security_info(market, stk_code)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='成交表';

-- 登录账户表
-- =========================
DROP TABLE IF EXISTS sys_account;
CREATE TABLE sys_account (
    account_id      BIGINT        NOT NULL AUTO_INCREMENT COMMENT '账户ID',
    username        VARCHAR(50)   NOT NULL COMMENT '用户名',
    password        VARCHAR(255)  NOT NULL COMMENT '密码（BCrypt加密）',
    role            VARCHAR(10)   NOT NULL COMMENT '角色(OPERATOR/CUSTOMER/ADMIN)',
    ref_id          BIGINT        DEFAULT NULL COMMENT '关联ID（客户填customer_code，操作员管理员填NULL）',
    status          VARCHAR(10)   NOT NULL DEFAULT '0' COMMENT '账户状态(0正常/1禁用)',
    created_time    DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_time    DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (account_id),
    UNIQUE KEY uk_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='登录账户表';
